💻 Developer Integration Guide

🐍 Python Quickstart

from master_compressor import MasterCompressor, compress_json, decompress_json
from master_compressor.database import SQLiteCompressedStore

# 1. High-Performance JSON & Data Compression
payload = {"user_id": 1024, "event": "ORDER_PLACED", "items": ["sku_1", "sku_2"]}
result = compress_json(payload)
print(f"Space Saved: {result.savings_ratio}% | Algorithm: {result.algorithm}")

# 2. Lossless Decompression with CRC32 Verification
restored = decompress_json(result.data)

# 3. Transparent Database Storage
db = SQLiteCompressedStore("analytics.db")
db.store("order_1024", payload)
order = db.retrieve("order_1024")
print("Database stats:", db.get_stats())
